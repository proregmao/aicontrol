❌ 发现多个严重问题

❌ 问题总结

核心问题：

  • 硬件检测硬编码：返回固定的模拟硬件信息，不是真实检测
  • 网络接口信息错误：使用硬编码的网络接口数据
  • 重启关机无效：只返回成功消息，没有真实执行SSH命令
  • 状态检测模拟：服务器状态使用固定模拟数据
  • 连接测试假实现：使用随机数模拟连接结果

1. 实现真实的SSH硬件检测

// 修复后的CPU检测实现 func (s *ServerService) getCPUInfoViaSSH(req models.ServerHardwareDetectRequest) (models.CPUInfo, error) { // 创建SSH连接 sshClient := ssh.NewSSHClient(req.IPAddress, req.Port, req.Username, req.Password, req.PrivateKey) if err := sshClient.Connect(); err != nil { return models.CPUInfo{}, err } defer sshClient.Disconnect() // 获取CPU型号 modelResult, err := sshClient.ExecuteCommand("cat /proc/cpuinfo | grep 'model name' | head -1 | cut -d':' -f2 | xargs") if err != nil { return models.CPUInfo{}, err } // 获取CPU核心数 coresResult, err := sshClient.ExecuteCommand("nproc") if err != nil { return models.CPUInfo{}, err } // 获取CPU使用率 usageResult, err := sshClient.ExecuteCommand("top -bn1 | grep 'Cpu(s)' | awk '{print $2}' | cut -d'%' -f1") if err != nil { return models.CPUInfo{}, err } // 解析结果 cores, _ := strconv.Atoi(strings.TrimSpace(coresResult.Output)) usage, _ := strconv.ParseFloat(strings.TrimSpace(usageResult.Output), 64) return models.CPUInfo{ Model: strings.TrimSpace(modelResult.Output), Cores: cores, Usage: usage, }, nil }

// 修复后的网络接口检测 func (s *ServerService) getNetworkInfoViaSSH(req models.ServerHardwareDetectRequest) ([]models.NetworkInfo, error) { sshClient := ssh.NewSSHClient(req.IPAddress, req.Port, req.Username, req.Password, req.PrivateKey) if err := sshClient.Connect(); err != nil { return nil, err } defer sshClient.Disconnect() // 获取网络接口列表 result, err := sshClient.ExecuteCommand("ip -o link show | awk -F': ' '{print $2}' | grep -v lo") if err != nil { return nil, err } var networks []models.NetworkInfo interfaces := strings.Split(strings.TrimSpace(result.Output), "\n") for _, iface := range interfaces { if iface == "" { continue } // 获取接口IP地址 ipResult, _ := sshClient.ExecuteCommand(fmt.Sprintf("ip addr show %s | grep 'inet ' | awk '{print $2}' | cut -d'/' -f1", iface)) // 获取MAC地址 macResult, _ := sshClient.ExecuteCommand(fmt.Sprintf("cat /sys/class/net/%s/address", iface)) // 获取接口状态 statusResult, _ := sshClient.ExecuteCommand(fmt.Sprintf("cat /sys/class/net/%s/operstate", iface)) networks = append(networks, models.NetworkInfo{ Name: iface, IP: strings.TrimSpace(ipResult.Output), MAC: strings.TrimSpace(macResult.Output), Status: strings.TrimSpace(statusResult.Output), Speed: "Unknown", // 需要额外命令获取 }) } return networks, nil }
